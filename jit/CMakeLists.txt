cmake_minimum_required(VERSION 3.24)
project(OceanBlastJit LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The initial native backend requires 64-bit Windows")
endif()
include(FetchContent)
set(ASMJIT_STATIC ON CACHE BOOL "" FORCE)
set(ASMJIT_TEST OFF CACHE BOOL "" FORCE)
set(ASMJIT_NO_FOREIGN ON CACHE BOOL "" FORCE)
set(ASMJIT_NO_UJIT ON CACHE BOOL "" FORCE)
# Pinned source; updates require an explicit dependency change and validation.
FetchContent_Declare(asmjit
  GIT_REPOSITORY https://github.com/asmjit/asmjit.git
  GIT_TAG 0464e1c8f1f43c861a0c73d4e59437fff255cdd9)
FetchContent_MakeAvailable(asmjit)
add_library(oceanblast_jit_core STATIC
  ../src/cpu/arm920t.cpp ../src/memory/bus.cpp ../src/cpu/x64_execution_engine.cpp)
target_compile_features(oceanblast_jit_core PUBLIC cxx_std_17)
target_include_directories(oceanblast_jit_core PUBLIC ../src)
target_link_libraries(oceanblast_jit_core PUBLIC asmjit::asmjit)
set_target_properties(oceanblast_jit_core PROPERTIES UNITY_BUILD ON UNITY_BUILD_BATCH_SIZE 8)
# GCC 16.2 fails internally when the AsmJit emitter joins this unity unit.
set_source_files_properties(../src/cpu/x64_execution_engine.cpp PROPERTIES SKIP_UNITY_BUILD_INCLUSION ON)
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  target_compile_options(oceanblast_jit_core PRIVATE -Wall -Wextra)
endif()
add_executable(x64_execution ../tests/x64_execution.cpp)
target_link_libraries(x64_execution PRIVATE oceanblast_jit_core)
enable_testing()
add_executable(x64_clocked_timing ../tests/x64_clocked_timing.cpp)
target_link_libraries(x64_clocked_timing PRIVATE oceanblast_jit_core)
add_test(NAME x64_clocked_timing COMMAND x64_clocked_timing)
add_test(NAME x64_execution COMMAND x64_execution)
add_executable(x64_entry_abi ../tests/x64_entry_abi.cpp)
target_link_libraries(x64_entry_abi PRIVATE oceanblast_jit_core)
add_test(NAME x64_entry_abi COMMAND x64_entry_abi)
add_executable(native_semantics ../tests/translation_frontend.cpp)
target_compile_definitions(native_semantics PRIVATE OCEANBLAST_TEST_NATIVE=1)
target_link_libraries(native_semantics PRIVATE oceanblast_jit_core)
add_test(NAME native_semantics COMMAND native_semantics)
add_executable(x64_memory ../tests/x64_memory.cpp)
target_link_libraries(x64_memory PRIVATE oceanblast_jit_core)
add_test(NAME x64_memory COMMAND x64_memory)
add_executable(x64_invalidation ../tests/x64_invalidation.cpp)
target_link_libraries(x64_invalidation PRIVATE oceanblast_jit_core)
add_test(NAME x64_invalidation COMMAND x64_invalidation)
add_executable(jit_cpu_regression ../tests/cpu_regression.cpp)
target_link_libraries(jit_cpu_regression PRIVATE oceanblast_jit_core)
add_test(NAME jit_cpu_regression COMMAND jit_cpu_regression)
add_executable(jit_emulation_timing ../tests/emulation_timing.cpp)
target_link_libraries(jit_emulation_timing PRIVATE oceanblast_jit_core)
add_test(NAME jit_emulation_timing COMMAND jit_emulation_timing)
set_tests_properties(jit_emulation_timing PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/..")
add_executable(timing_contract ../tests/timing_contract.cpp)
target_link_libraries(timing_contract PRIVATE oceanblast_jit_core)
add_test(NAME timing_contract COMMAND timing_contract)
foreach(device_check audio_clock timer4 dma_audio_stream)
  add_executable(timing_${device_check} ../tests/${device_check}.cpp)
  target_link_libraries(timing_${device_check} PRIVATE oceanblast_jit_core)
  add_test(NAME timing_${device_check} COMMAND timing_${device_check})
endforeach()
find_package(Python3 QUIET COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_test(NAME timing_landmarks COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../tests/test_timing_landmarks.py")
endif()

enable_language(RC)
# Resource paths are relative to this generated file, independent of build cwd.
file(TO_CMAKE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../assets" OCEANBLAST_ASSETS)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/oceanblast.rc"
  "101 RCDATA \"${OCEANBLAST_ASSETS}/oceanblast-logo.png\"\n102 RCDATA \"${OCEANBLAST_ASSETS}/oceanblast-symbol.png\"\n")
add_executable(oceanblast_jit_player
  "${CMAKE_CURRENT_BINARY_DIR}/oceanblast.rc"
  ../src/main.cpp ../src/cartridge/cart_parser.cpp
  ../src/display/display_win32.cpp ../src/audio/audio_win32.cpp)
target_compile_definitions(oceanblast_jit_player PRIVATE OCEANBLAST_WITH_JIT=1)
target_link_libraries(oceanblast_jit_player PRIVATE oceanblast_jit_core
  gdi32 user32 winmm comdlg32 comctl32 shell32 ole32 dinput8 dxguid d3d11 dxgi d3dcompiler gdiplus)

set(OCEANBLAST_PGO_MODE "OFF" CACHE STRING "GCC profile mode: OFF, GENERATE or USE")
set_property(CACHE OCEANBLAST_PGO_MODE PROPERTY STRINGS OFF GENERATE USE)
set(OCEANBLAST_PGO_DIRECTORY "" CACHE PATH "Matching GCC profile directory")
if(NOT OCEANBLAST_PGO_MODE STREQUAL "OFF")
  if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU" OR NOT OCEANBLAST_PGO_DIRECTORY)
    message(FATAL_ERROR "Profile builds require GCC and OCEANBLAST_PGO_DIRECTORY")
  endif()
  if(OCEANBLAST_PGO_MODE STREQUAL "GENERATE")
    set(OCEANBLAST_PGO_FLAG "-fprofile-generate=${OCEANBLAST_PGO_DIRECTORY}")
  elseif(OCEANBLAST_PGO_MODE STREQUAL "USE")
    set(OCEANBLAST_PGO_FLAG "-fprofile-use=${OCEANBLAST_PGO_DIRECTORY}")
    target_compile_options(oceanblast_jit_core PRIVATE -Werror=coverage-mismatch -Werror=missing-profile)
    target_compile_options(oceanblast_jit_player PRIVATE -Werror=coverage-mismatch -Werror=missing-profile)
  else()
    message(FATAL_ERROR "Unknown OCEANBLAST_PGO_MODE")
  endif()
  # Keep the same build tree and object paths between generation and use.
  # Only project code is trained; the pinned emitter library is unchanged.
  target_compile_options(oceanblast_jit_core PRIVATE "${OCEANBLAST_PGO_FLAG}")
  target_compile_options(oceanblast_jit_player PRIVATE "${OCEANBLAST_PGO_FLAG}")
  target_link_options(oceanblast_jit_core INTERFACE "${OCEANBLAST_PGO_FLAG}")
endif()
